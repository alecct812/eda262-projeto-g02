            -- foi mais descumprida entre jan/2017 e ago/2018, considerando a quantidade e a taxa
            -- de pedidos entregues apos a data prevista?
            SELECT
              uf_cliente,
              count(*)                                               AS pedidos_elegiveis,
              count_if(atrasado)                                     AS pedidos_atrasados,
