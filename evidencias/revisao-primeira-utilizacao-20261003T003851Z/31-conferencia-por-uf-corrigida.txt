import csv, json
m = json.load(open("parte-1/dados/trusted/manifesto.json", encoding="utf-8"))["pergunta_referencia"]["por_uf"]
ath = {r["uf_cliente"]: r for r in csv.DictReader(open("evidencias/manual/consulta-resultado.csv", encoding="utf-8"))}
dif = 0
for uf, loc in sorted(m.items()):
    a = ath.get(uf)
    taxa = round(100 * loc["atrasados"] / loc["elegiveis"], 2)
    ok = (a is not None and int(a["pedidos_elegiveis"]) == loc["elegiveis"]
          and int(a["pedidos_atrasados"]) == loc["atrasados"]
          and abs(float(a["taxa_atraso_pct"]) - taxa) < 0.006)
    dif += not ok
    print(f"{uf}: athena {a['pedidos_elegiveis']} elegiveis, {a['pedidos_atrasados']} atrasados, {a['taxa_atraso_pct']}% | "
          f"local {loc['elegiveis']}, {loc['atrasados']}, {taxa}% | {'OK' if ok else 'DIFERENTE'}")
print("UFs no Athena:", len(ath), "| UFs no calculo local:", len(m), "| divergencias:", dif)
raise SystemExit(1 if dif or len(ath) != len(m) else 0)
